   890→// Reusable GenericForm
   891→function GenericForm({ initial, fields, onSave }: { initial?: Record<string, any>; fields: { name: string; label: string; type?: string; required?: boolean }[]; onSave: (data: Record<string, any>) => Promise<void> }) {
   892→  const [data, setData] = useState<Record<string, any>>(initial || {});
   893→  const { toast } = useToast();
   894→
   895→  const handleSave = async () => {
   896→    for (const f of fields) {
   897→      if (f.required && !data[f.name]) {
   898→        toast({ title: `Le champ "${f.label}" est obligatoire`, variant: 'destructive' });
   899→        return;
   900→      }
   901→    }
   902→    await onSave(data);
   903→    toast({ title: 'Sauvegardé avec succès' });
   904→  };
   905→
   906→  return (
   907→    <>
   908→      <DialogHeader><DialogTitle>{initial ? 'Modifier' : 'Ajouter'}</DialogTitle></DialogHeader>
   909→      <div className="space-y-4 pt-4">
   910→        {fields.map((field) => (
   911→          <div key={field.name} className="space-y-2">
   912→            <Label>{field.label} {field.required && <span className="text-red-500">*</span>}</Label>
   913→            {field.type === 'textarea' ? (
   914→              <Textarea value={data[field.name] || ''} onChange={(e) => setData({ ...data, [field.name]: e.target.value })} rows={3} />
   915→            ) : (
   916→              <Input type={field.type === 'number' ? 'number' : 'text'} value={data[field.name] || ''} onChange={(e) => setData({ ...data, [field.name]: e.target.value })} />
   917→            )}
   918→          </div>
   919→        ))}
   920→        <Button onClick={handleSave} className="bg-gdv-gold hover:bg-gdv-gold-light text-white w-full">
   921→          <Save className="w-4 h-4 mr-2" /> Sauvegarder
   922→        </Button>
   923→      </div>
   924→    </>
   925→  );
   926→}

   928→// Service form
   929→function ServiceForm({ initial, onSave }: { initial?: Record<string, any>; onSave: (data: Record<string, any>) => Promise<void> }) {
   930→  const [data, setData] = useState<Record<string, any>>(initial || { icon: 'Plane', visible: true });
   931→  const { toast } = useToast();
   932→
   933→  const handleSave = async () => {
   934→    if (!data.title || !data.slug) {
   935→      toast({ title: 'Titre et slug sont obligatoires', variant: 'destructive' });
   936→      return;
   937→    }
   938→    const features = data.featuresText ? data.featuresText.split('\n').filter(Boolean) : (data.features || []);
   939→    await onSave({ ...data, features: JSON.stringify(features) });
   940→    toast({ title: 'Service sauvegardé' });
   941→  };
   942→
   943→  return (
   944→    <>
   945→      <DialogHeader><DialogTitle>{initial ? 'Modifier le service' : 'Ajouter un service'}</DialogTitle></DialogHeader>
   946→      <div className="space-y-4 pt-4">
   947→        <div className="grid grid-cols-2 gap-4">
   948→          <div className="space-y-2"><Label>Titre *</Label><Input value={data.title || ''} onChange={(e) => setData({ ...data, title: e.target.value })} /></div>
   949→          <div className="space-y-2"><Label>Slug *</Label><Input value={data.slug || ''} onChange={(e) => setData({ ...data, slug: e.target.value.toLowerCase().replace(/\s+/g, '-').replace(/[^a-z0-9-]/g, '') })} /></div>
