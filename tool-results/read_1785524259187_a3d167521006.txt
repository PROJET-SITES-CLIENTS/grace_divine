   891→   891→function GenericForm({ initial, fields, onSave }: { initial?: Record<string, any>; fields: { name: string; label: string; type?: string; required?: boolean }[]; onSave: (data: Record<string, any>) => Promise<void> }) {
   892→   892→  const [data, setData] = useState<Record<string, any>>(initial || {});
   893→   893→  const { toast } = useToast();
   894→   894→
   895→   895→  const handleSave = async () => {
   896→   896→    for (const f of fields) {
   897→   897→      if (f.required && !data[f.name]) {
   898→   898→        toast({ title: `Le champ "${f.label}" est obligatoire`, variant: 'destructive' });
   899→   899→        return;
   900→   900→      }
   901→   901→    }
   902→   902→    await onSave(data);
   903→   903→    toast({ title: 'Sauvegardé avec succès' });
   904→   904→  };
   905→   905→
   906→   906→  return (
   907→   907→    <>
   908→   908→      <DialogHeader><DialogTitle>{initial ? 'Modifier' : 'Ajouter'}</DialogTitle></DialogHeader>
   909→   909→      <div className="space-y-4 pt-4">
   910→   910→        {fields.map((field) => (
   911→   911→          <div key={field.name} className="space-y-2">
   912→   912→            <Label>{field.label} {field.required && <span className="text-red-500">*</span>}</Label>
   913→   913→            {field.type === 'textarea' ? (
   914→   914→              <Textarea value={data[field.name] || ''} onChange={(e) => setData({ ...data, [field.name]: e.target.value })} rows={3} />
   915→   915→            ) : (
   916→   916→              <Input type={field.type === 'number' ? 'number' : 'text'} value={data[field.name] || ''} onChange={(e) => setData({ ...data, [field.name]: e.target.value })} />
   917→   917→            )}
   918→   918→          </div>
   919→   919→        ))}
   920→   920→        <Button onClick={handleSave} className="bg-gdv-gold hover:bg-gdv-gold-light text-white w-full">
   921→   921→          <Save className="w-4 h-4 mr-2" /> Sauvegarder
   922→   922→        </Button>
   923→   923→      </div>
   924→   924→    </>
   925→   925→  );
   926→   926→}
