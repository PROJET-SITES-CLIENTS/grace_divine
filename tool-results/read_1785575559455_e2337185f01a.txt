   101→    { key: 'settings', label: 'Contenu du Site', icon: <Settings className="w-5 h-5" /> },

   103→    { key: 'services', label: 'Services', icon: <Plane className="w-5 h-5" /> },

   225→        <div className="p-4 border-b border-gdv-gold/20">

   346→            {/* Settings */}

   352→                    <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   353→                      <div className="space-y-2">

   380→                      </div>

   388→                      </div>
   389→                      <div className="space-y-2">
   390→                        <Label>Facebook URL</Label>

   901→// Reusable GenericForm
   902→function GenericForm({ initial, fields, onSave }: { initial?: Record<string, any>; fields: { name: string; label: string; type?: string; required?: boolean }[]; onSave: (data: Record<string, any>) => Promise<void> }) {
   903→  const [data, setData] = useState<Record<string, any>>(initial || {});
   904→  const { toast } = useToast();
   905→
   906→  const handleSave = async () => {
   907→    for (const f of fields) {
   908→      if (f.required && !data[f.name]) {
   909→        toast({ title: `Le champ "${f.label}" est obligatoire`, variant: 'destructive' });
   910→        return;
   911→      }
   912→    }
   913→    await onSave(data);
   914→    toast({ title: 'Sauvegardé avec succès' });
   915→  };
   916→
   917→  return (
   918→    <>
   919→      <DialogHeader><DialogTitle>{initial ? 'Modifier' : 'Ajouter'}</DialogTitle></DialogHeader>
   920→      <div className="space-y-4 pt-4">
   921→        {fields.map((field) => (
   922→          <div key={field.name} className="space-y-2">
   923→            <Label>{field.label} {field.required && <span className="text-red-500">*</span>}</Label>
   924→            {field.type === 'textarea' ? (
   925→              <Textarea value={data[field.name] || ''} onChange={(e) => setData({ ...data, [field.name]: e.target.value })} rows={3} />
   926→            ) : (
   927→              <Input type={field.type === 'number' ? 'number' : 'text'} value={data[field.name] || ''} onChange={(e) => setData({ ...data, [field.name]: e.target.value })} />
   928→            )}
   929→          </div>
   930→        ))}
   931→        <Button onClick={handleSave} className="bg-gdv-teal hover:bg-gdv-teal-light text-white w-full">
   932→          <Save className="w-4 h-4 mr-2" /> Sauvegarder
   933→        </Button>
   934→      </div>
   935→    </>
   936→  );
   937→}

   939→// Service form
   940→function ServiceForm({ initial, onSave }: { initial?: Record<string, any>; onSave: (data: Record<string, any>) => Promise<void> }) {
   941→  const [data, setData] = useState<Record<string, any>>(initial || { icon: 'Plane', visible: true });
   942→  const { toast } = useToast();
   943→
   944→  const handleSave = async () => {
   945→    if (!data.title || !data.slug) {
   946→      toast({ title: 'Titre et slug sont obligatoires', variant: 'destructive' });
   947→      return;
   948→    }
   949→    const features = data.featuresText ? data.featuresText.split('\n').filter(Boolean) : (data.features || []);
   950→    await onSave({ ...data, features: JSON.stringify(features) });
   951→    toast({ title: 'Service sauvegardé' });
   952→  };
   953→
   954→  return (
   955→    <>
   956→      <DialogHeader><DialogTitle>{initial ? 'Modifier le service' : 'Ajouter un service'}</DialogTitle></DialogHeader>
   957→      <div className="space-y-4 pt-4">
   958→        <div className="grid grid-cols-2 gap-4">
   959→          <div className="space-y-2"><Label>Titre *</Label><Input value={data.title || ''} onChange={(e) => setData({ ...data, title: e.target.value })} /></div>
   960→          <div className="space-y-2"><Label>Slug *</Label><Input value={data.slug || ''} onChange={(e) => setData({ ...data, slug: e.target.value.toLowerCase().replace(/\s+/g, '-').replace(/[^a-z0-9-]/g, '') })} /></div>
   961→        </div>

   963→        <div className="space-y-2"><Label>Courte description</Label><Input value={data.shortDesc || ''} onChange={(e) => setData({ ...data, shortDesc: e.target.value })} /></div>
   964→        <div className="space-y-2"><Label>Description complète</Label><Textarea value={data.description || ''} onChange={(e) => setData({ ...data, description: e.target.value })} rows={4} /></div>
   965→        <div className="space-y-2">
   966→          <Label>Fonctionnalités (une par ligne)</Label>
   967→          <Textarea value={data.featuresText || (Array.isArray(data.features) ? data.features.join('\n') : '')} onChange={(e) => setData({ ...data, featuresText: e.target.value })} rows={4} placeholder="Vols domestiques et internationaux&#10;Comparaison de prix&#10;Options flexibles" />
   968→        </div>
   969→        <Button onClick={handleSave} className="bg-gdv-teal hover:bg-gdv-teal-light text-white w-full"><Save className="w-4 h-4 mr-2" />Sauvegarder</Button>
   970→      </div>
   971→    </>
   972→  );
