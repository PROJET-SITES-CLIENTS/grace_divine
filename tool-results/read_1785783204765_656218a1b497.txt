   225→    toast({ title: 'Élément créé' });

   228→  };

   362→                      </TableBody>

   364→                  </CardContent>
   365→                </Card>
   366→              </div>
   367→            )}
   368→
   369→            {/* Settings */}
   370→            {activeTab === 'settings' && (
   371→              <div className="max-w-3xl">
   372→                <h2 className="text-2xl font-bold text-gray-900 mb-6">Contenu du Site</h2>

   387→                      </div>

   390→                        <Input value={settings.phone2 || ''} onChange={(e) => setSettings({ ...settings, phone2: e.target.value })} />
   391→                      </div>
   392→                      <div className="space-y-2">
   393→                        <Label>Email 1</Label>
   394→                        <Input value={settings.email1 || ''} onChange={(e) => setSettings({ ...settings, email1: e.target.value })} />
   395→                      </div>
   396→                      <div className="space-y-2">
   397→                        <Label>Email 2</Label>
   398→                        <Input value={settings.email2 || ''} onChange={(e) => setSettings({ ...settings, email2: e.target.value })} />
   399→                      </div>
   400→                      <div className="space-y-2 sm:col-span-2">
   401→                        <Label>Adresse</Label>
   402→                        <Input value={settings.address || ''} onChange={(e) => setSettings({ ...settings, address: e.target.value })} />
   403→                      </div>
   404→                      <div className="space-y-2">
   405→                        <Label>Numéro WhatsApp</Label>
   406→                        <Input value={settings.whatsappNumber || ''} onChange={(e) => setSettings({ ...settings, whatsappNumber: e.target.value })} />
   407→                      </div>

   411→                      </div>

   413→                        <Label>Facebook URL</Label>

   415→                      </div>

   419→                      </div>
   420→                      <div className="space-y-2">
   421→                        <Label>Twitter URL</Label>
   422→                        <Input value={settings.twitterUrl || ''} onChange={(e) => setSettings({ ...settings, twitterUrl: e.target.value })} />
   423→                      </div>
   424→                      <div className="space-y-2">

   426→                        <Input value={settings.tiktokUrl || ''} onChange={(e) => setSettings({ ...settings, tiktokUrl: e.target.value })} />

   430→                        <Input value={settings.linkedinUrl || ''} onChange={(e) => setSettings({ ...settings, linkedinUrl: e.target.value })} />
   431→                      </div>

   924→// Reusable GenericForm
   925→function GenericForm({ initial, fields, onSave }: { initial?: Record<string, any>; fields: { name: string; label: string; type?: string; required?: boolean }[]; onSave: (data: Record<string, any>) => Promise<void> }) {
   926→  const [data, setData] = useState<Record<string, any>>(initial || {});
   927→  const { toast } = useToast();
   928→
   929→  const handleSave = async () => {
   930→    for (const f of fields) {
   931→      if (f.required && !data[f.name]) {
   932→        toast({ title: `Le champ "${f.label}" est obligatoire`, variant: 'destructive' });
   933→        return;
   934→      }
   935→    }
   936→    await onSave(data);
   937→    toast({ title: 'Sauvegardé avec succès' });
   938→  };
   939→
   940→  return (
   941→    <>
   942→      <DialogHeader><DialogTitle>{initial ? 'Modifier' : 'Ajouter'}</DialogTitle></DialogHeader>
   943→      <div className="space-y-4 pt-4">
   944→        {fields.map((field) => (
   945→          <div key={field.name} className="space-y-2">
   946→            <Label>{field.label} {field.required && <span className="text-red-500">*</span>}</Label>
   947→            {field.type === 'textarea' ? (
   948→              <Textarea value={data[field.name] || ''} onChange={(e) => setData({ ...data, [field.name]: e.target.value })} rows={3} />
   949→            ) : (
   950→              <Input type={field.type === 'number' ? 'number' : 'text'} value={data[field.name] || ''} onChange={(e) => setData({ ...data, [field.name]: e.target.value })} />
   951→            )}
   952→          </div>
   953→        ))}
   954→        <Button onClick={handleSave} className="bg-gdv-teal hover:bg-gdv-teal-light text-white w-full">
   955→          <Save className="w-4 h-4 mr-2" /> Sauvegarder
   956→        </Button>
   957→      </div>
   958→    </>
   959→  );
   960→}

   962→// Service form
   963→function ServiceForm({ initial, onSave }: { initial?: Record<string, any>; onSave: (data: Record<string, any>) => Promise<void> }) {
   964→  const [data, setData] = useState<Record<string, any>>(initial || { icon: 'Plane', visible: true });
   965→  const { toast } = useToast();
   966→
   967→  const handleSave = async () => {
   968→    if (!data.title || !data.slug) {
   969→      toast({ title: 'Titre et slug sont obligatoires', variant: 'destructive' });
   970→      return;
   971→    }
   972→    const features = data.featuresText ? data.featuresText.split('\n').filter(Boolean) : (data.features || []);
   973→    await onSave({ ...data, features: JSON.stringify(features) });
   974→    toast({ title: 'Service sauvegardé' });
   975→  };
   976→
   977→  return (
   978→    <>
   979→      <DialogHeader><DialogTitle>{initial ? 'Modifier le service' : 'Ajouter un service'}</DialogTitle></DialogHeader>
   980→      <div className="space-y-4 pt-4">
   981→        <div className="grid grid-cols-2 gap-4">
   982→          <div className="space-y-2"><Label>Titre *</Label><Input value={data.title || ''} onChange={(e) => setData({ ...data, title: e.target.value })} /></div>
   983→          <div className="space-y-2"><Label>Slug *</Label><Input value={data.slug || ''} onChange={(e) => setData({ ...data, slug: e.target.value.toLowerCase().replace(/\s+/g, '-').replace(/[^a-z0-9-]/g, '') })} /></div>

   986→        <div className="space-y-2"><Label>Courte description</Label><Input value={data.shortDesc || ''} onChange={(e) => setData({ ...data, shortDesc: e.target.value })} /></div>
   987→        <div className="space-y-2"><Label>Description complète</Label><Textarea value={data.description || ''} onChange={(e) => setData({ ...data, description: e.target.value })} rows={4} /></div>
   988→        <div className="space-y-2">
   989→          <Label>Fonctionnalités (une par ligne)</Label>
   990→          <Textarea value={data.featuresText || (Array.isArray(data.features) ? data.features.join('\n') : '')} onChange={(e) => setData({ ...data, featuresText: e.target.value })} rows={4} placeholder="Vols domestiques et internationaux&#10;Comparaison de prix&#10;Options flexibles" />
   991→        </div>
   992→        <Button onClick={handleSave} className="bg-gdv-teal hover:bg-gdv-teal-light text-white w-full"><Save className="w-4 h-4 mr-2" />Sauvegarder</Button>
   993→      </div>
