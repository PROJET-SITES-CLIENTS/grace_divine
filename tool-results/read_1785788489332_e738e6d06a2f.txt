   367→                  </CardContent>
   368→                </Card>

   378→                    <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   379→                      <div className="space-y-2">

   386→                      </div>

   390→                      </div>
   391→                      <div className="space-y-2">
   392→                        <Label>Téléphone 2</Label>
   393→                        <Input value={settings.phone2 || ''} onChange={(e) => setSettings({ ...settings, phone2: e.target.value })} />
   394→                      </div>

   396→                        <Label>Email 1</Label>

   398→                      </div>

   406→                      </div>

   416→                        <Label>Facebook URL</Label>

   422→                      </div>

   927→// Reusable GenericForm
   928→function GenericForm({ initial, fields, onSave }: { initial?: Record<string, any>; fields: { name: string; label: string; type?: string; required?: boolean }[]; onSave: (data: Record<string, any>) => Promise<void> }) {
   929→  const [data, setData] = useState<Record<string, any>>(initial || {});
   930→  const { toast } = useToast();
   931→
   932→  const handleSave = async () => {
   933→    for (const f of fields) {
   934→      if (f.required && !data[f.name]) {
   935→        toast({ title: `Le champ "${f.label}" est obligatoire`, variant: 'destructive' });
   936→        return;
   937→      }
   938→    }
   939→    await onSave(data);
   940→    toast({ title: 'Sauvegardé avec succès' });
   941→  };
   942→
   943→  return (
   944→    <>
   945→      <DialogHeader><DialogTitle>{initial ? 'Modifier' : 'Ajouter'}</DialogTitle></DialogHeader>
   946→      <div className="space-y-4 pt-4">
   947→        {fields.map((field) => (
   948→          <div key={field.name} className="space-y-2">
   949→            <Label>{field.label} {field.required && <span className="text-red-500">*</span>}</Label>
   950→            {field.type === 'textarea' ? (
   951→              <Textarea value={data[field.name] || ''} onChange={(e) => setData({ ...data, [field.name]: e.target.value })} rows={3} />
   952→            ) : (
   953→              <Input type={field.type === 'number' ? 'number' : 'text'} value={data[field.name] || ''} onChange={(e) => setData({ ...data, [field.name]: e.target.value })} />
   954→            )}
   955→          </div>
   956→        ))}
   957→        <Button onClick={handleSave} className="bg-gdv-teal hover:bg-gdv-teal-light text-white w-full">
   958→          <Save className="w-4 h-4 mr-2" /> Sauvegarder
   959→        </Button>
   960→      </div>
   961→    </>
   962→  );
   963→}

   966→function ServiceForm({ initial, onSave }: { initial?: Record<string, any>; onSave: (data: Record<string, any>) => Promise<void> }) {
   967→  const [data, setData] = useState<Record<string, any>>(initial || { icon: 'Plane', visible: true });
   968→  const { toast } = useToast();

   970→  const handleSave = async () => {
   971→    if (!data.title || !data.slug) {
   972→      toast({ title: 'Titre et slug sont obligatoires', variant: 'destructive' });
   973→      return;
   974→    }

   981→    <>
   982→      <DialogHeader><DialogTitle>{initial ? 'Modifier le service' : 'Ajouter un service'}</DialogTitle></DialogHeader>

   984→        <div className="grid grid-cols-2 gap-4">
   985→          <div className="space-y-2"><Label>Titre *</Label><Input value={data.title || ''} onChange={(e) => setData({ ...data, title: e.target.value })} /></div>
