   928→   928→function GenericForm({ initial, fields, onSave }: { initial?: Record<string, any>; fields: { name: string; label: string; type?: string; required?: boolean }[]; onSave: (data: Record<string, any>) => Promise<void> }) {
   929→   929→  const [data, setData] = useState<Record<string, any>>(initial || {});
   930→   930→  const { toast } = useToast();
   931→   931→
   932→   932→  const handleSave = async () => {
   933→   933→    for (const f of fields) {
   934→   934→      if (f.required && !data[f.name]) {
   935→   935→        toast({ title: `Le champ "${f.label}" est obligatoire`, variant: 'destructive' });
   936→   936→        return;
   937→   937→      }
   938→   938→    }
   939→   939→    await onSave(data);
   940→   940→    toast({ title: 'Sauvegardé avec succès' });
   941→   941→  };
   942→   942→
   943→   943→  return (
   944→   944→    <>
   945→   945→      <DialogHeader><DialogTitle>{initial ? 'Modifier' : 'Ajouter'}</DialogTitle></DialogHeader>
   946→   946→      <div className="space-y-4 pt-4">
   947→   947→        {fields.map((field) => (
   948→   948→          <div key={field.name} className="space-y-2">
   949→   949→            <Label>{field.label} {field.required && <span className="text-red-500">*</span>}</Label>
   950→   950→            {field.type === 'textarea' ? (
   951→   951→              <Textarea value={data[field.name] || ''} onChange={(e) => setData({ ...data, [field.name]: e.target.value })} rows={3} />
   952→   952→            ) : (
   953→   953→              <Input type={field.type === 'number' ? 'number' : 'text'} value={data[field.name] || ''} onChange={(e) => setData({ ...data, [field.name]: e.target.value })} />
   954→   954→            )}
   955→   955→          </div>
   956→   956→        ))}
   957→   957→        <Button onClick={handleSave} className="bg-gdv-teal hover:bg-gdv-teal-light text-white w-full">
   958→   958→          <Save className="w-4 h-4 mr-2" /> Sauvegarder
   959→   959→        </Button>
   960→   960→      </div>
   961→   961→    </>
   962→   962→  );
   963→   963→}
