   935→// Reusable GenericForm
   936→function GenericForm({ initial, fields, onSave }: { initial?: Record<string, any>; fields: { name: string; label: string; type?: string; required?: boolean }[]; onSave: (data: Record<string, any>) => Promise<void> }) {
   937→  const [data, setData] = useState<Record<string, any>>(initial || {});
   938→  const { toast } = useToast();
   939→
   940→  const handleSave = async () => {
   941→    for (const f of fields) {
   942→      if (f.required && !data[f.name]) {
   943→        toast({ title: `Le champ "${f.label}" est obligatoire`, variant: 'destructive' });
   944→        return;
   945→      }
   946→    }
   947→    await onSave(data);
   948→    toast({ title: 'Sauvegardé avec succès' });
   949→  };
   950→
   951→  return (
   952→    <>
   953→      <DialogHeader><DialogTitle>{initial ? 'Modifier' : 'Ajouter'}</DialogTitle></DialogHeader>
   954→      <div className="space-y-4 pt-4">
   955→        {fields.map((field) => (
   956→          <div key={field.name} className="space-y-2">
   957→            <Label>{field.label} {field.required && <span className="text-red-500">*</span>}</Label>
   958→            {field.type === 'textarea' ? (
   959→              <Textarea value={data[field.name] || ''} onChange={(e) => setData({ ...data, [field.name]: e.target.value })} rows={3} />
   960→            ) : (
   961→              <Input type={field.type === 'number' ? 'number' : 'text'} value={data[field.name] || ''} onChange={(e) => setData({ ...data, [field.name]: e.target.value })} />
   962→            )}
   963→          </div>
   964→        ))}
   965→        <Button onClick={handleSave} className="bg-gdv-teal hover:bg-gdv-teal-light text-white w-full">
   966→          <Save className="w-4 h-4 mr-2" /> Sauvegarder
   967→        </Button>
   968→      </div>
   969→    </>
   970→  );
   971→}

   974→function ServiceForm({ initial, onSave }: { initial?: Record<string, any>; onSave: (data: Record<string, any>) => Promise<void> }) {
   975→  const [data, setData] = useState<Record<string, any>>(initial || { icon: 'Plane', visible: true });

   981→      return;
   982→    }

   994→          <div className="space-y-2"><Label>Slug *</Label><Input value={data.slug || ''} onChange={(e) => setData({ ...data, slug: e.target.value.toLowerCase().replace(/\s+/g, '-').replace(/[^a-z0-9-]/g, '') })} /></div>
